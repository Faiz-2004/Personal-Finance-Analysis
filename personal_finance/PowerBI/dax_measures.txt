// -------------------------------------------------------------------------------
// 1. CORE FINANCIAL MEASURES
// -------------------------------------------------------------------------------

// 1.1 Total Income
// Calculates the sum of all cash inflow transactions
Total Income = 
CALCULATE(
    SUM(Transactions[Amount]),
    Transactions[Type] = "Income"
)

// 1.2 Total Expenses
// Calculates the sum of all cash outflow transactions
Total Expenses = 
CALCULATE(
    SUM(Transactions[Amount]),
    Transactions[Type] = "Expense"
)

// 1.3 Total Savings
// Net liquidity retained from total earnings
Total Savings = 
[Total Income] - [Total Expenses]

// 1.4 Savings Rate (%)
// Percentage of gross income saved
Savings Rate = 
DIVIDE([Total Savings], [Total Income], 0) * 100

// 1.5 Average Monthly Expenses
// Evaluates monthly baseline expenditure using the Calendar table
Average Monthly Expenses = 
AVERAGEX(
    VALUES('Calendar'[MonthYear]),
    [Total Expenses]
)

// 1.6 Total Budget
// Sum of planned allocations from monthly_budget
Total Budget = 
SUM(Monthly_Budget[Budget_Amount])

// 1.7 Budget Variance
// Positive value indicates savings under budget; negative indicates cost overrun
Budget Variance = 
[Total Budget] - [Total Expenses]

// 1.8 Budget Variance %
// Relative percentage difference from the allocated budget
Budget Variance % = 
DIVIDE([Budget Variance], [Total Budget], 0) * 100


// -------------------------------------------------------------------------------
// 2. SUPPORTING & FORMATTING MEASURES
// -------------------------------------------------------------------------------

// 2.1 Total Transactions Count
Total Transactions Count = 
COUNTROWS(Transactions)

// 2.2 Expense Transactions Count
Expense Transactions Count = 
CALCULATE(
    COUNTROWS(Transactions),
    Transactions[Type] = "Expense"
)

// 2.3 Income Transactions Count
Income Transactions Count = 
CALCULATE(
    COUNTROWS(Transactions),
    Transactions[Type] = "Income"
)

// 2.4 Average Expense Per Transaction (Ticket Size)
Avg Expense Ticket = 
DIVIDE([Total Expenses], [Expense Transactions Count], 0)

// 2.5 Dynamic KPI Card Title / Subtitle
Savings Status Callout = 
IF(
    [Savings Rate] >= 20,
    "Healthy Savings (> 20%)",
    "Low Savings (< 20% Benchmark)"
)

// 2.6 Budget Health Flag (for conditional formatting colors)
// Returns 1 for Within Budget (Green), -1 for Over Budget (Red)
Budget Health Indicator = 
IF([Budget Variance] >= 0, 1, -1)


// -------------------------------------------------------------------------------
// 3. CALENDAR TABLE DAX DEFINITION (Calculated Table)
// -------------------------------------------------------------------------------
// In Power BI Desktop: Click 'Modeling' > 'New Table' and paste the following:

Calendar = 
ADDCOLUMNS(
    CALENDAR(DATE(2026, 1, 1), DATE(2026, 12, 31)),
    "Year", YEAR([Date]),
    "Month Number", MONTH([Date]),
    "Month", FORMAT([Date], "MMMM"),
    "Month Short", FORMAT([Date], "MMM"),
    "MonthYear", FORMAT([Date], "yyyy-MM"),
    "Quarter", "Q" & FORMAT([Date], "Q")
)

// Note: Sort 'Month' column by 'Month Number' to ensure chronological order.
